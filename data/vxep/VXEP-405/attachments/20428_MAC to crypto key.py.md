"""
This script converts a MAC address to a crypto key.  The MAC address should be in the format of XX:XX:XX:XX:XX:XX

Note that this script only generates the first 5 of 8 keys.  The remaining two keys are generated by the autopilot and
cannot be predicted by this script.
"""

mac_address = input("Enter MAC address: ")
key_prefix = "0x"
key_separator = " "
default_key = "0xaa55aa55"

def mac_to_list(mac_address):
    """
    Returns a list of hex values from a MAC address
    :param mac_address:
    :return:
    """
    mac_list = mac_address.split(":")
    return mac_list


def hex_xor(a, b):
    """
    Returns the XOR of two hex values
    :param a: Value A
    :param b: Value B
    :return: Xor of A and B
    """
    hex_value = hex(int(a, 16) ^ int(b, 16))
    # Drop the 0x
    hex_value = hex_value[2:]
    return hex_value

def generate_key_0(mac_address):
    """
    Generates the first key line
    :param mac_address:
    :return: Key 0
    """

    key_0_list = mac_to_list(mac_address)[0:4]
    # flip the order of the list
    key_0_list.reverse()

    # Generate the key
    key_0 = key_prefix + "".join(key_0_list)
    return key_0


def generate_key_1(mac_address):
    """
    Generates the second key line.  First 4 characters are XOR of last two segments of MAC address and "FF".
    last 4 characters are the last two segments of the MAC address
    :param mac_address:
    :return: Key 1
    """
    # Generate the first 4 characters of the key
    key_1_list = mac_to_list(mac_address)[0:2]
    #Flip the order of the list
    key_1_list.reverse()

    # XOR the contents of the list
    key_1_list[0] = hex_xor(key_1_list[0], "FF")
    key_1_list[1] = hex_xor(key_1_list[1], "FF")

    # Generate the last 4 characters of the key
    key_1_list_second = mac_to_list(mac_address)[4:6]
    # flip the order of the list
    key_1_list_second.reverse()

    # Generate the key
    key_1 = key_prefix + "".join(key_1_list) + "".join(key_1_list_second)
    return key_1


def generate_key_2(mac_address):
    """
    Generates the third key line.  This key is an XOR of the first 4 segments of the MAC address and "FF"
    :param mac_address:
    :return: Key 2
    """
    key_2_list = mac_to_list(mac_address)[2:6]
    # flip the order of the list
    key_2_list.reverse()

    # XOR the contents of the list
    key_2_list[0] = hex_xor(key_2_list[0], "FF")
    key_2_list[1] = hex_xor(key_2_list[1], "FF")
    key_2_list[2] = hex_xor(key_2_list[2], "FF")
    key_2_list[3] = hex_xor(key_2_list[3], "FF")

    # Generate the key
    key_2 = key_prefix + "".join(key_2_list)
    return key_2

# Generate a full key
key = generate_key_0(mac_address) + key_separator + \
      generate_key_1(mac_address) + key_separator + \
      generate_key_2(mac_address) + key_separator + \
      default_key + key_separator + \
      default_key + key_separator + \
      default_key + key_separator

print(key)
